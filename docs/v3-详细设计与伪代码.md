## 三、状态定义 — state/ProcessState.ts

```typescript
import { Token, TraceEntry, ProcessDefinition } from '../types';

// ============================================================
// 3.1 流程实例状态
// ============================================================

export type ProcessStatus = 'running' | 'completed' | 'terminated';

export interface ProcessState {
  /** 流程实例唯一 ID */
  id: string;
  /** 流程名称 */
  name: string;
  /** 流程状态 */
  status: ProcessStatus;
  /** 关联的流程定义 ID */
  definitionId: string;
  /** 流程级共享变量 */
  variables: Record<string, any>;
  /** 当前所有令牌（活跃 + 挂起） */
  tokens: Token[];
  /** 执行轨迹 */
  trace: TraceEntry[];
  /** 并行网关汇聚等待表：{ gatewayId: [已到达的 tokenId] } */
  _gatewayWait: Record<string, string[]>;
  /** 时间戳 */
  createdAt: Date;
  startedAt: Date;
  endedAt?: Date;
}

// ============================================================
// 3.2 序列化 / 反序列化
// ============================================================

export function serialize(state: ProcessState): string {
  // Date → ISO string
  return JSON.stringify(state, (key, value) => {
    if (value instanceof Date) return value.toISOString();
    return value;
  });
}

export function deserialize(json: string): ProcessState {
  const raw = JSON.parse(json);
  // 恢复 Date 字段
  raw.createdAt = new Date(raw.createdAt);
  raw.startedAt = new Date(raw.startedAt);
  if (raw.endedAt) raw.endedAt = new Date(raw.endedAt);
  raw.tokens = raw.tokens.map((t: any) => ({
    ...t,
    createdAt: new Date(t.createdAt),
    suspendedAt: t.suspendedAt ? new Date(t.suspendedAt) : undefined,
  }));
  raw.trace = raw.trace.map((e: any) => ({
    ...e,
    timestamp: new Date(e.timestamp),
  }));
  return raw;
}
```

```typescript
export interface NodeHookContext {
  /** 当前流程状态（只读快照） */

  token: Readonly<Token>;

  /** 当前节点完整信息 */

  /** 流程定义 */

   * 🔑 挂起当前令牌

   * 仅在 onNodeEnter 中有效

  /**
   * 合并数据到令牌的 data 字段
   */
  setTokenData(data: Record<string, any>): void;

  /**
   * 合并数据到流程级 variables
   */
  setVariables(variables: Record<string, any>): void;

  /**
   * 执行脚本（仅当注册了 ScriptExecutorPlugin 时可用）

   * @returns Promise<Record<string, any>> - 脚本执行结果

}
```

## 五、钩子管理器 — hooks/HookManager.ts

```typescript
import { HookEvent, HookHandlerMap, FlowEngineOptions } from './types';

/**
 * 钩子管理器
 * 从 FlowEngineOptions 中提取钩子函数，统一管理触发
 * 不再支持运行时动态注册——所有钩子在构造时确定
 */
export class HookManager {
  private handlers: Map<string, Function | undefined> = new Map();

  /**
   * 从 options 中提取并注册所有钩子
   */
  constructor(options: FlowEngineOptions = {}) {
    this.handlers.set('nodeEnter',     options.onNodeEnter);
    this.handlers.set('nodeLeave',     options.onNodeLeave);
    this.handlers.set('sequenceFlow',  options.onSequenceFlow);
    this.handlers.set('processStart',  options.onProcessStart);
    this.handlers.set('processEnd',    options.onProcessEnd);
  }

  /**
   * 触发钩子（支持 async）
   * 如果 handler 未注册，静默跳过
   * 如果 handler 抛异常，记录日志但不中断引擎
   */
  async emit<E extends HookEvent>(
    event: E,
    context: Parameters<HookHandlerMap[E]>[0]
  ): Promise<void> {
    const handler = this.handlers.get(event);
    if (!handler) return;

    try {
      await handler(context);
    } catch (error) {
      console.error(`[HookManager] ${event} handler error:`, error);
    }
  }
}
```

## 六、令牌管理器 — engine/TokenManager.ts

```typescript
import { ProcessState } from '../state/ProcessState';
import { Token } from '../types';
import { generateId } from '../state/IdGenerator';

export class TokenManager {

  /**
   * 在指定节点创建一个 active 令牌
   */
  createToken(
    state: ProcessState,
    elementId: string,
    data: Record<string, any> = {}
  ): ProcessState {
    const newToken: Token = {
      id: generateId(),
      elementId,
      status: 'active',
      data: { ...data },
      createdAt: new Date(),
    };

    return {
      ...state,
      tokens: [...state.tokens, newToken],
    };
  }

  /**
   * 挂起令牌（active → suspended）
   */
  suspendToken(state: ProcessState, tokenId: string): ProcessState {
    return {
      ...state,
      tokens: state.tokens.map(t =>
        t.id === tokenId
          ? { ...t, status: 'suspended' as const, suspendedAt: new Date() }
          : t
      ),
    };
  }

  /**
   * 恢复令牌（suspended → active），并可选合并数据
   */
  resumeToken(
    state: ProcessState,
    tokenId: string,
    data?: Record<string, any>
  ): ProcessState {
    return {
      ...state,
      tokens: state.tokens.map(t =>
        t.id === tokenId
          ? {
              ...t,
              status: 'active' as const,
              data: data ? { ...t.data, ...data } : t.data,
              suspendedAt: undefined,
            }
          : t
      ),
    };
  }

  /**
   * 销毁令牌（从列表中移除）
   */
  destroyToken(state: ProcessState, tokenId: string): ProcessState {
    return {
      ...state,
      tokens: state.tokens.filter(t => t.id !== tokenId),
    };
  }

  /**
   * 移动令牌到另一个节点（原地更新 elementId）
   */
  moveToken(
    state: ProcessState,
    tokenId: string,
    targetElementId: string
  ): ProcessState {
    return {
      ...state,
      tokens: state.tokens.map(t =>
        t.id === tokenId
          ? { ...t, elementId: targetElementId }
          : t
      ),
    };
  }

  /**
   * 更新令牌数据
   */
  updateTokenData(
    state: ProcessState,
    tokenId: string,
    data: Record<string, any>
  ): ProcessState {
    return {
      ...state,
      tokens: state.tokens.map(t =>
        t.id === tokenId
          ? { ...t, data: { ...t.data, ...data } }
          : t
      ),
    };
  }
}
```

## 七、网关解析器 — engine/GatewayResolver.ts

```typescript
import { ProcessState } from '../state/ProcessState';
import { Token, Element, ProcessDefinition } from '../types';
import { TokenManager } from './TokenManager';
import { evaluateExpression } from '../utils/ExpressionEvaluator';
import { generateId } from '../state/IdGenerator';

/**
 * 网关解析器
 * 引擎内部唯一保留的"节点执行逻辑"
 * 只处理网关类型，不处理任何业务节点
 */
export class GatewayResolver {
  private tokenManager: TokenManager;

  constructor(tokenManager: TokenManager) {
    this.tokenManager = tokenManager;
  }

  // ============================================================
  // 分发入口
  // ============================================================

  resolve(
    state: ProcessState,
    token: Token,
    gateway: Element,
    definition: ProcessDefinition
  ): ProcessState {
    switch (gateway.type) {
      case 'bpmn:exclusiveGateway':
        return this.resolveExclusive(state, token, gateway, definition);
      case 'bpmn:parallelGateway':
        return this.resolveParallel(state, token, gateway, definition);
      case 'bpmn:inclusiveGateway':
        return this.resolveInclusive(state, token, gateway, definition);
      default:
        // 未知网关类型，当作普通通过节点
        return this.passThrough(state, token, gateway, definition);
    }
  }

  // ============================================================
  // 排他网关（XOR）
  // ============================================================
  //
  // 规则：
  //   1. 按 outgoing 顺序逐条评估 conditionExpression
  //   2. 第一条为 true 的 flow 胜出
  //   3. 都不满足 → 走 default flow
  //   4. 没有 default → 报错
  //
  // 伪代码流程：
  //   销毁当前 token
  //   for flow in outgoing（跳过 default）:
  //     if evaluate(flow.condition, variables + tokenData) == true:
  //       在 flow.targetRef 创建新 token → return
  //   走 default flow → return
  //   无路可走 → 抛错
  // ============================================================

  private resolveExclusive(
    state: ProcessState,
    token: Token,
    gateway: Element,
    definition: ProcessDefinition
  ): ProcessState {
    const outgoing = gateway.outgoing || [];
    const defaultFlowId = gateway.properties?.default || null;
    const context = { ...state.variables, ...token.data };

    // 1. 销毁当前令牌
    state = this.tokenManager.destroyToken(state, token.id);

    // 2. 按顺序评估条件
    for (const flowId of outgoing) {
      if (flowId === defaultFlowId) continue;  // default 最后兜底

      const flow = definition.sequenceFlows.get(flowId);
      if (!flow) continue;

      // 无条件的 flow 视为 true
      if (!flow.conditionExpression) {
        return this.tokenManager.createToken(state, flow.targetRef, token.data);
      }

      if (evaluateExpression(flow.conditionExpression, context)) {
        return this.tokenManager.createToken(state, flow.targetRef, token.data);
      }
    }

    // 3. 走 default
    if (defaultFlowId) {
      const defaultFlow = definition.sequenceFlows.get(defaultFlowId);
      if (defaultFlow) {
        return this.tokenManager.createToken(state, defaultFlow.targetRef, token.data);
      }
    }

    // 4. 只有一条出边时，无条件走
    if (outgoing.length === 1) {
      const flow = definition.sequenceFlows.get(outgoing[0]);
      if (flow) {
        return this.tokenManager.createToken(state, flow.targetRef, token.data);
      }
    }

    // 5. 无路可走
    throw new Error(
      `排他网关 ${gateway.id} 无满足条件的分支且无默认路径`
    );
  }

  // ============================================================
  // 并行网关（AND）
  // ============================================================
  //
  // 两种模式根据 incoming/outgoing 数量自动判断：
  //
  // 【分裂模式】 incoming <= 1, outgoing > 1
  //   销毁当前 token
  //   for each outgoing flow:
  //     在 flow.targetRef 创建新 token
  //
  // 【汇聚模式】 incoming > 1
  //   将当前 token 记录到 _gatewayWait[gatewayId]
  //   销毁当前 token
  //   if 已到达数量 == incoming 数量:
  //     清空 _gatewayWait[gatewayId]
  //     for each outgoing flow:
  //       在 flow.targetRef 创建新 token
  //   else:
  //     等待（不创建新 token）
  // ============================================================

  private resolveParallel(
    state: ProcessState,
    token: Token,
    gateway: Element,
    definition: ProcessDefinition
  ): ProcessState {
    const incoming = gateway.incoming || [];
    const outgoing = gateway.outgoing || [];

    if (incoming.length <= 1) {
      // ---- 分裂模式 ----
      state = this.tokenManager.destroyToken(state, token.id);
      for (const flowId of outgoing) {
        const flow = definition.sequenceFlows.get(flowId);
        if (flow) {
          state = this.tokenManager.createToken(state, flow.targetRef, token.data);
        }
      }
      return state;
    }

    // ---- 汇聚模式 ----
    // 记录到达
    const waitKey = gateway.id;
    const arrived = [...(state._gatewayWait[waitKey] || []), token.id];
    state = {
      ...state,
      _gatewayWait: { ...state._gatewayWait, [waitKey]: arrived },
    };

    // 销毁当前令牌
    state = this.tokenManager.destroyToken(state, token.id);

    // 检查是否全部到达
    if (arrived.length >= incoming.length) {
      // 全部到达，清空等待，向下游发令牌
      const { [waitKey]: _, ...restWait } = state._gatewayWait;
      state = { ...state, _gatewayWait: restWait };

      for (const flowId of outgoing) {
        const flow = definition.sequenceFlows.get(flowId);
        if (flow) {
          state = this.tokenManager.createToken(state, flow.targetRef, token.data);
        }
      }
    }
    // 否则：等待其他分支的令牌到达

    return state;
  }

  // ============================================================
  // 包含网关（OR）
  // ============================================================
  //
  // 【分裂模式】
  //   与排他网关类似，但所有满足条件的分支都走
  //   for flow in outgoing:
  //     if evaluate(flow.condition) == true:
  //       在 flow.targetRef 创建新 token
  //   如果没有任何满足 → 走 default
  //
  // 【汇聚模式】
  //   与并行网关类似，但不需要等所有 incoming
  //   只等待"实际激活的"分支到达
  //   （简化实现：等待所有 incoming，和并行网关一样）
  // ============================================================

  private resolveInclusive(
    state: ProcessState,
    token: Token,
    gateway: Element,
    definition: ProcessDefinition
  ): ProcessState {
    const incoming = gateway.incoming || [];
    const outgoing = gateway.outgoing || [];

    if (incoming.length <= 1) {
      // ---- 分裂模式 ----
      const context = { ...state.variables, ...token.data };
      const defaultFlowId = gateway.properties?.default || null;
      state = this.tokenManager.destroyToken(state, token.id);

      let anySelected = false;

      for (const flowId of outgoing) {
        if (flowId === defaultFlowId) continue;
        const flow = definition.sequenceFlows.get(flowId);
        if (!flow) continue;

        // 无条件 → 选中
        if (!flow.conditionExpression ||
            evaluateExpression(flow.conditionExpression, context)) {
          state = this.tokenManager.createToken(state, flow.targetRef, token.data);
          anySelected = true;
        }
      }

      // 都不满足 → default
      if (!anySelected && defaultFlowId) {
        const defaultFlow = definition.sequenceFlows.get(defaultFlowId);
        if (defaultFlow) {
          state = this.tokenManager.createToken(state, defaultFlow.targetRef, token.data);
        }
      }

      return state;
    }

    // ---- 汇聚模式（简化：同并行网关） ----
    return this.resolveParallel(state, token, gateway, definition);
  }

  // ============================================================
  // 直通（兜底）
  // ============================================================

  private passThrough(
    state: ProcessState,
    token: Token,
    node: Element,
    definition: ProcessDefinition
  ): ProcessState {
    const outgoing = node.outgoing || [];
    state = this.tokenManager.destroyToken(state, token.id);

    for (const flowId of outgoing) {
      const flow = definition.sequenceFlows.get(flowId);
      if (flow) {
        state = this.tokenManager.createToken(state, flow.targetRef, token.data);
      }
    }
    return state;
  }
}
```

## 八、连线遍历器 — engine/FlowTraverser.ts

```typescript
import { ProcessState } from '../state/ProcessState';
import { Token, Element, SequenceFlow, ProcessDefinition, TraceEntry } from '../types';
import { TokenManager } from './TokenManager';
import { HookManager } from '../hooks/HookManager';

/**
 * 连线遍历器
 * 负责：令牌离开一个节点 → 沿 SequenceFlow → 到达下一个节点
 * 每经过一条 SequenceFlow 都会触发 onSequenceFlow 钩子
 */
export class FlowTraverser {
  private tokenManager: TokenManager;
  private hooks: HookManager;

  constructor(tokenManager: TokenManager, hooks: HookManager) {
    this.tokenManager = tokenManager;
    this.hooks = hooks;
  }

  /**
   * 将令牌从当前节点推进到所有出边的目标节点
   *
   * 步骤：
   *   1. 读取 currentNode.outgoing（出边列表）
   *   2. 销毁当前 token
   *   3. 对每条出边：
   *      a. 查 definition.sequenceFlows 获取 flow 对象
   *      b. 查 definition.elements 获取 targetNode
   *      c. 触发 onSequenceFlow 钩子
   *      d. 记录 trace
   *      e. 在 targetNode 创建新 token
   *   4. 返回新 state
   */
  async traverse(
    state: ProcessState,
    token: Token,
    currentNode: Element,
    definition: ProcessDefinition
  ): Promise<ProcessState> {
    const outgoing = currentNode.outgoing || [];

    // 没有出边 → 销毁令牌（死路）
    if (outgoing.length === 0) {
      return this.tokenManager.destroyToken(state, token.id);
    }

    // 销毁当前令牌
    state = this.tokenManager.destroyToken(state, token.id);

    // 遍历每条出边
    for (const flowId of outgoing) {
      const flow = definition.sequenceFlows.get(flowId);
      if (!flow) continue;

      const targetNode = definition.elements.get(flow.targetRef);
      if (!targetNode) continue;

      // 触发 sequenceFlow 钩子
      await this.hooks.emit('sequenceFlow', {
        state,
        token,
        flow,
        sourceNode: currentNode,
        targetNode,
        definition,
      });

      // 记录 trace
      state = this.addTrace(state, {
        type: 'sequence-flow',
        elementId: flow.id,
        elementType: 'sequenceFlow',
        elementName: flow.name,
        tokenId: token.id,
        timestamp: new Date(),
        sourceRef: flow.sourceRef,
        targetRef: flow.targetRef,
        flowId: flow.id,
      });

      // 在目标节点创建新令牌
      state = this.tokenManager.createToken(state, flow.targetRef, token.data);
    }

    return state;
  }

  /**
   * 添加 trace 记录
   */
  private addTrace(state: ProcessState, entry: TraceEntry): ProcessState {
    return {
      ...state,
      trace: [...state.trace, entry],
    };
  }
}
```

## 九、主引擎 — engine/FlowEngine.ts（核心）

```typescript
import { ProcessState, ProcessStatus } from '../state/ProcessState';
import { Token, Element, ProcessDefinition, TraceEntry } from '../types';
import { TokenManager } from './TokenManager';
import { GatewayResolver } from './GatewayResolver';
import { FlowTraverser } from './FlowTraverser';
import { HookManager } from '../hooks/HookManager';
import {
  NodeHookContext, FlowHookContext, HookEvent, HookHandlerMap,
  FlowEngineOptions, ScriptExecutorPlugin,
} from '../hooks/types';
import BPMNParser from '../parser/BPMNParser';
import { evaluateExpressionResult } from '../utils/ExpressionEvaluator';
import { generateId } from '../state/IdGenerator';

// ============================================================
// 辅助：判断节点类别
// ============================================================

const GATEWAY_TYPES = new Set([
  'bpmn:exclusiveGateway',
  'bpmn:parallelGateway',
  'bpmn:inclusiveGateway',
  'bpmn:eventBasedGateway',
]);

function isGateway(type: string): boolean {
  return GATEWAY_TYPES.has(type);
}

function isEndEvent(type: string): boolean {
  return type === 'bpmn:endEvent';
}

function isStartEvent(type: string): boolean {
  return type === 'bpmn:startEvent';
}

// ============================================================
// 插件接口
// ============================================================

export interface FlowPlugin {
  name: string;
  version: string;
  install(engine: FlowEngine): void;
}

// ============================================================
// 主引擎
// ============================================================

export class FlowEngine {
  private tokenManager: TokenManager;
  private gatewayResolver: GatewayResolver;
  private flowTraverser: FlowTraverser;
  private hooks: HookManager;
  private scriptExecutor: ScriptExecutorPlugin | null;

  /**
   * 构造函数 — 所有钩子和插件在此注入
   *
   * @example
   * const engine = new FlowEngine({
   *   onNodeEnter: (ctx) => {
   *     if (ctx.node.type === 'bpmn:userTask') ctx.suspend();
   *   },
   *   onNodeLeave: (ctx) => { console.log('离开', ctx.node.name); },
   *   onSequenceFlow: (ctx) => { console.log(ctx.flow.id); },
   *   scriptExecutor: myScriptPlugin,
   * });
   */
  constructor(options: FlowEngineOptions = {}) {
    this.tokenManager = new TokenManager();
    this.hooks = new HookManager(options);
    this.gatewayResolver = new GatewayResolver(this.tokenManager);
    this.flowTraverser = new FlowTraverser(this.tokenManager, this.hooks);
    this.scriptExecutor = options.scriptExecutor || null;
  }

  // ============================================================
  // 公开 API：生命周期
  // ============================================================

  /**
   * 启动流程
   *
   * 伪代码：
   *   1. parse(bpmnXML) → definition
   *   2. 创建初始 state
   *   3. 触发 processStart 钩子
   *   4. 找到 startEvent，创建 token
   *   5. 进入推进循环 advance()
   *   6. 返回 state（可能有 suspended tokens）
   */
  async startProcess(
    bpmnXML: string,
    initialData: Record<string, any> = {}
  ): Promise<ProcessState> {

    // 1. 解析 BPMN
    const definition = BPMNParser.parse(bpmnXML);

    // 2. 创建初始状态
    const now = new Date();
    let state: ProcessState = {
      id: generateId(),
      name: definition.name || definition.id,
      status: 'running',
      definitionId: definition.id,
      variables: { ...initialData },
      tokens: [],
      trace: [],
      _gatewayWait: {},
      createdAt: now,
      startedAt: now,
    };

    // 3. processStart 钩子
    await this.hooks.emit('processStart', { state, definition });

    // 4. 找到 startEvent
    const startEvent = this.findStartEvent(definition);
    state = this.tokenManager.createToken(state, startEvent.id, {});

    // 5. 推进循环
    state = await this.advance(state, definition);

    return state;
  }

  /**
   * 恢复挂起的令牌
   *
   * 伪代码：
   *   1. 找到 suspended token
   *   2. 合并 data
   *   3. token.status → active
   *   4. 触发 nodeLeave 钩子
   *   5. 沿出边移动到下一个节点（FlowTraverser）
   *   6. 进入推进循环 advance()
   *   7. 返回新 state
   *
   * 注意：调用方需要持有最新的 state 和原始 bpmnXML
   *       因为引擎是无状态的，不存储任何东西
   */
  async resume(
    state: ProcessState,
    tokenId: string,
    bpmnXML: string,
    data?: Record<string, any>
  ): Promise<ProcessState> {

    // 1. 解析定义
    const definition = BPMNParser.parse(bpmnXML);

    // 2. 找到并恢复令牌
    const token = state.tokens.find(t => t.id === tokenId);
    if (!token) {
      throw new Error(`Token ${tokenId} not found`);
    }
    if (token.status !== 'suspended') {
      throw new Error(`Token ${tokenId} is not suspended (status: ${token.status})`);
    }

    state = this.tokenManager.resumeToken(state, tokenId, data);

    // 3. 获取令牌当前所在节点
    const node = definition.elements.get(token.elementId);
    if (!node) {
      throw new Error(`Node ${token.elementId} not found in definition`);
    }

    // 4. 触发 nodeLeave 钩子
    const updatedToken = state.tokens.find(t => t.id === tokenId)!;
    await this.emitNodeLeave(state, updatedToken, node, definition);

    // 5. trace: node-leave
    state = this.addTrace(state, {
      type: 'node-leave',
      elementId: node.id,
      elementType: node.type,
      elementName: node.name,
      tokenId: tokenId,
      timestamp: new Date(),
    });

    // 6. 沿出边移动
    state = await this.flowTraverser.traverse(
      state, updatedToken, node, definition
    );

    // 7. 推进循环
    state = await this.advance(state, definition);

    return state;
  }

  /**
   * 终止流程
   */
  terminate(state: ProcessState): ProcessState {
    return {
      ...state,
      status: 'terminated',
      tokens: [],  // 销毁所有令牌
      endedAt: new Date(),
    };
  }

  // ============================================================
  // 公开 API：查询
  // ============================================================

  /**
   * 获取所有挂起的令牌信息
   * 外部应用用这个知道"哪些节点在等我处理"
   */
  getSuspendedTokens(
    state: ProcessState,
    bpmnXML: string
  ): SuspendedTokenInfo[] {
    const definition = BPMNParser.parse(bpmnXML);
    return state.tokens
      .filter(t => t.status === 'suspended')
      .map(t => {
        const node = definition.elements.get(t.elementId);
        return {
          tokenId: t.id,
          nodeId: t.elementId,
          nodeType: node?.type || 'unknown',
          nodeName: node?.name || '',
          nodeProperties: node?.properties || {},
          tokenData: t.data,
          suspendedAt: t.suspendedAt!,
        };
      });
  }

  // ============================================================
  // 公开 API：插件
  // ============================================================

  use(plugin: FlowPlugin): this {
    plugin.install(this);
    return this;
  }

  // ============================================================
  // 私有：推进循环（引擎的心脏）
  // ============================================================

  /**
   * advance() — 核心推进循环
   *
   * 反复扫描所有 active token，直到：
   *   a. 没有 active token（全部 suspended 或已销毁）
   *   b. 达到最大迭代次数（防死循环）
   *
   * 每个 active token 的处理：
   *
   *   ┌─────────────────────────────────────────────────┐
   *   │ 读取 token.elementId → 查 definition 得到 node  │
   *   └──────────┬──────────────────────────────────────┘
   *              │
   *              ▼
   *   ┌──────────────────┐
   *   │ node.type 是什么？│
   *   └──┬───┬───┬───┬───┘
   *      │   │   │   │
   *      │   │   │
   *      │   │   └─ 其他节点（含 scriptTask）──→ emitNodeEnter()
   *      │   │                                   │
   *      │   │                              调了 suspend?
   *      │   │                             /          \
   *      │   │                          是             否
   *      │   │                           │              │
   *      │   │                     token 停住      emitNodeLeave()
   *      │   │                     等 resume()     flowTraverser.traverse()
   *      │   │                                     → 到下一个节点
   *      │   │
   *      │   │   scriptTask 在 onNodeEnter 中可调 ctx.executeScript()
   *      │   │   使用 ScriptExecutorPlugin 执行脚本
   *      │   │
   *      │   └─ endEvent ──→ 销毁 token，记 trace
   *      │
   *      └─ gateway ──→ emitNodeEnter()（suspend 无效）
   *      │                gatewayResolver.resolve()
   *      │
   *      └─ startEvent ──→ emitNodeLeave()
   *                         flowTraverser.traverse()
   *                         → 自动走到第一个节点
   */
  private async advance(
    state: ProcessState,
    definition: ProcessDefinition
  ): Promise<ProcessState> {

    let hasProgress = true;
    let iterations = 0;
    const MAX = 1000;

    while (hasProgress && iterations < MAX) {
      hasProgress = false;
      iterations++;

      // 获取所有 active token
      const activeTokens = state.tokens.filter(t => t.status === 'active');
      if (activeTokens.length === 0) break;

      // 逐个处理
      for (const token of activeTokens) {
        // 检查 token 是否还在（可能被前面的处理销毁了）
        if (!state.tokens.some(t => t.id === token.id && t.status === 'active')) {
          continue;
        }

        const node = definition.elements.get(token.elementId);
        if (!node) continue;

        // ---- 分类处理 ----

        if (node.type === 'bpmn:startEvent') {
          // StartEvent：记录 trace，自动通过
          state = this.addTrace(state, {
            type: 'node-enter',
            elementId: node.id,
            elementType: node.type,
            elementName: node.name,
            tokenId: token.id,
            timestamp: new Date(),
          });

          await this.emitNodeEnter(state, token, node, definition, false);

          state = this.addTrace(state, {
            type: 'node-leave',
            elementId: node.id,
            elementType: node.type,
            elementName: node.name,
            tokenId: token.id,
            timestamp: new Date(),
          });

          await this.emitNodeLeave(state, token, node, definition);
          state = await this.flowTraverser.traverse(state, token, node, definition);
          hasProgress = true;

        } else if (isEndEvent(node.type)) {
          // EndEvent：记录 trace，销毁令牌
          state = this.addTrace(state, {
            type: 'node-enter',
            elementId: node.id,
            elementType: node.type,
            elementName: node.name,
            tokenId: token.id,
            timestamp: new Date(),
          });

          await this.emitNodeEnter(state, token, node, definition, false);

          state = this.tokenManager.destroyToken(state, token.id);

          state = this.addTrace(state, {
            type: 'node-leave',
            elementId: node.id,
            elementType: node.type,
            elementName: node.name,
            tokenId: token.id,
            timestamp: new Date(),
          });

          await this.emitNodeLeave(state, token, node, definition);
          hasProgress = true;

        } else if (isGateway(node.type)) {
          // Gateway：触发钩子（suspend 无效），引擎内部处理
          state = this.addTrace(state, {
            type: 'node-enter',
            elementId: node.id,
            elementType: node.type,
            elementName: node.name,
            tokenId: token.id,
            timestamp: new Date(),
          });

          // 触发 nodeEnter（suspend 无效）
          await this.emitNodeEnter(state, token, node, definition, false);

          // 网关逻辑
          state = this.gatewayResolver.resolve(state, token, node, definition);

          state = this.addTrace(state, {
            type: 'gateway-resolve',
            elementId: node.id,
            elementType: node.type,
            elementName: node.name,
            tokenId: token.id,
            timestamp: new Date(),
          });

          hasProgress = true;

        } else {
          // ---- 普通业务节点（含 scriptTask）：钩子决定一切 ----

          state = this.addTrace(state, {
            type: 'node-enter',
            elementId: node.id,
            elementType: node.type,
            elementName: node.name,
            tokenId: token.id,
            timestamp: new Date(),
          });

          // 触发 nodeEnter（suspend 有效）
          let suspended = false;

          const ctx = this.buildNodeHookContext(
            state, token, node, definition,
            /* suspendFn */ () => { suspended = true; }
          );
          await this.hooks.emit('nodeEnter', ctx);

          // 应用 ctx 中的数据变更
          state = this.applyContextChanges(state, token.id, ctx);

          if (suspended) {
            // 外部调用了 suspend() → 挂起令牌
            state = this.tokenManager.suspendToken(state, token.id);
            // 这个 token 不再推进，等 resume()
          } else {
            // 外部没有调 suspend() → 自动通过
            state = this.addTrace(state, {
              type: 'node-leave',
              elementId: node.id,
              elementType: node.type,
              elementName: node.name,
              tokenId: token.id,
              timestamp: new Date(),
            });

            await this.emitNodeLeave(state, token, node, definition);
            state = await this.flowTraverser.traverse(state, token, node, definition);
            hasProgress = true;
          }
        }
      }
    }

    // 检查流程是否结束
    state = this.checkCompletion(state, definition);

    return state;
  }

  // ============================================================
  // 私有：钩子触发辅助
  // ============================================================

  /**
   * 构建 NodeHookContext
   */
  private buildNodeHookContext(
    state: ProcessState,
    token: Token,
    node: Element,
    definition: ProcessDefinition,
    suspendFn: () => void
  ): NodeHookContext & { _tokenDataPatch: Record<string, any>; _varPatch: Record<string, any> } {
    const _tokenDataPatch: Record<string, any> = {};
    const _varPatch: Record<string, any> = {};
    const scriptExecutor = this.scriptExecutor;

    return {
      state,
      token,
      node,
      definition,
      suspend: suspendFn,
      setTokenData: (data) => Object.assign(_tokenDataPatch, data),
      setVariables: (vars) => Object.assign(_varPatch, vars),

      /**
       * 执行脚本 — 调用 ScriptExecutorPlugin
       * 外部应用在 onNodeEnter 中调用：
       *   const result = await ctx.executeScript(ctx.node.properties.script);
       *   ctx.setTokenData(result);
       */
      executeScript: (script: string): Promise<Record<string, any>> => {
        if (!scriptExecutor) {
          return Promise.reject(new Error(
            'executeScript 不可用：未注册 ScriptExecutorPlugin'
          ));
        }
        return new Promise((resolve, reject) => {
          try {
            scriptExecutor.execute(state, script, (error, result) => {
              if (error) reject(error);
              else resolve(result || {});
            });
          } catch (e) {
            reject(e);
          }
        });
      },

      _tokenDataPatch,
      _varPatch,
    };
  }

  /**
   * 将钩子中 setTokenData / setVariables 的变更应用到 state
   */
  private applyContextChanges(
    state: ProcessState,
    tokenId: string,
    ctx: { _tokenDataPatch: Record<string, any>; _varPatch: Record<string, any> }
  ): ProcessState {
    if (Object.keys(ctx._tokenDataPatch).length > 0) {
      state = this.tokenManager.updateTokenData(state, tokenId, ctx._tokenDataPatch);
    }
    if (Object.keys(ctx._varPatch).length > 0) {
      state = { ...state, variables: { ...state.variables, ...ctx._varPatch } };
    }
    return state;
  }

  /**
   * 触发 nodeEnter 钩子（suspend 可配置）
   */
  private async emitNodeEnter(
    state: ProcessState,
    token: Token,
    node: Element,
    definition: ProcessDefinition,
    allowSuspend: boolean
  ): Promise<void> {
    await this.hooks.emit('nodeEnter', {
      state,
      token,
      node,
      definition,
      suspend: allowSuspend ? () => {} : () => {},
      setTokenData: () => {},
      setVariables: () => {},
    });
  }

  /**
   * 触发 nodeLeave 钩子
   */
  private async emitNodeLeave(
    state: ProcessState,
    token: Token,
    node: Element,
    definition: ProcessDefinition
  ): Promise<void> {
    await this.hooks.emit('nodeLeave', {
      state,
      token,
      node,
      definition,
      suspend: () => {},
      setTokenData: () => {},
      setVariables: () => {},
    });
  }

  // ============================================================
  // 私有：辅助方法
  // ============================================================

  private findStartEvent(definition: ProcessDefinition): Element {
    for (const [, element] of definition.elements) {
      if (element.type === 'bpmn:startEvent') {
        return element;
      }
    }
    throw new Error('流程定义中未找到 startEvent');
  }

  private checkCompletion(
    state: ProcessState,
    definition: ProcessDefinition
  ): ProcessState {
    // 没有任何令牌 → 流程结束
    if (state.tokens.length === 0 && state.status === 'running') {
      state = {
        ...state,
        status: 'completed',
        endedAt: new Date(),
      };
      // 异步触发 processEnd 钩子（fire and forget）
      this.hooks.emit('processEnd', { state, definition });
    }
    return state;
  }

  private addTrace(state: ProcessState, entry: TraceEntry): ProcessState {
    return {
      ...state,
      trace: [...state.trace, entry],
    };
  }
}

// ============================================================
// 导出类型
// ============================================================

export interface SuspendedTokenInfo {
  tokenId: string;
  nodeId: string;
  nodeType: string;
  nodeName: string;
  nodeProperties: Record<string, any>;
  tokenData: Record<string, any>;
  suspendedAt: Date;
}
```

```
  │                          │─────────────────>│            │              │            │
  │                          │                  │            │              │            │
  │                          │ ═══ advance() 循环开始 ═══    │              │            │
  │                          │                  │            │              │            │
  │                          │ token在startEvent             │              │            │
  │                          │ → 自动通过                     │              │            │
  │                          │─────────────────────────────────────────────>│            │
  │                          │ emit('nodeEnter', startEvent)               │            │
  │                          │─────────────────────────────────────────────────────────>│
  │                          │─────────────────────────────────────────────>│            │
  │                          │ emit('nodeLeave', startEvent)               │            │
  │                          │─────────────────────────────────────────────────────────>│
  │                          │                  │            │              │            │
  │                          │ traverse(startEvent→flow→nodeA)             │            │
  │                          │────────────────────────────────────────────>│            │
  │                          │                  │            │  emit('sequenceFlow')     │
  │                          │                  │            │              │───────────>│
  │                          │                  │            │  createToken(nodeA)       │
  │                          │                  │<───────────│──────────────│            │
  │                          │                  │            │              │            │
  │                          │ token在nodeA（假设是userTask）│              │            │
  │                          │─────────────────────────────────────────────────────────>│
  │                          │ emit('nodeEnter', userTask)                  │            │
  │                          │                  │            │              │   外部调了  │
  │                          │                  │            │              │  suspend() │
  │                          │                  │            │              │            │
  │                          │ suspendToken(tokenId)         │              │            │
  │                          │─────────────────>│            │              │            │
  │                          │                  │            │              │            │
  │                          │ ═══ advance() 循环结束（无 active token）═══ │            │
  │                          │                  │            │              │            │
  │  <── return state ───────│                  │            │              │            │
  │  (含 1 个 suspended token)                  │            │              │            │
```

### 12.1 请假审批流程（完整可运行伪代码）

```typescript
import {
  FlowEngine, ProcessState, NodeHookContext,
  FlowEngineOptions, ScriptExecutorPlugin
} from 'bpmn-flash';

// ========== 持久化层（你自己实现） ==========
const db = new MyDatabase();

// ========== 脚本执行插件（保留自 v2.x） ==========
const myScriptPlugin: ScriptExecutorPlugin = {
  execute(state, script, cb) {
    try {
      // 你自己的安全脚本执行逻辑
      const sandbox = createSandbox(state.variables);
      const result = sandbox.run(script);
      cb(null, result);
    } catch (error) {
      cb(error as Error);
    }
  }
};

// ========== 创建引擎（所有钩子在构造时传入） ==========
const engine = new FlowEngine({

  // --- 脚本插件：bpmn:scriptTask 由引擎自动执行 ---
  scriptExecutor: myScriptPlugin,

  // --- 节点进入钩子 ---
  onNodeEnter: async (ctx: NodeHookContext) => {
    const { node, token, state } = ctx;

    switch (node.type) {
      // ---- 用户任务：挂起，等人审批 ----
      case 'bpmn:userTask': {
        ctx.suspend();

        // 写入你自己的任务表
        await db.tasks.create({
          processId: state.id,
          tokenId: token.id,
          nodeId: node.id,
          nodeName: node.name,
          assignee: node.properties.assignee || null,
          candidateGroups: node.properties.candidateGroups || null,
          formData: token.data,
          status: 'pending',
          createdAt: new Date(),
        });

        // 发通知
        await notifyService.send(node.properties.assignee, {
          title: `待办：${node.name}`,
          link: `/approve/${state.id}/${token.id}`,
        });
        break;
      }

      // ---- 服务任务：挂起，调 API ----
      case 'bpmn:serviceTask': {
        ctx.suspend();

        // 异步调 API，完成后在外部调 engine.resume()
        callExternalAPI(node.properties.serviceUrl, token.data)
          .then(result => {
            // resume 在另一个调用栈中完成
            return resumeFromService(state.id, token.id, result);
          })
          .catch(error => {
            return resumeFromService(state.id, token.id, { error: error.message });
          });
        break;
      }

      // ---- 发送任务：发邮件/消息，不挂起 ----
      case 'bpmn:sendTask': {
        await emailService.send({
          to: node.properties.recipient,
          subject: node.properties.subject,
          body: renderTemplate(node.properties.template, token.data),
        });
        // 不调 suspend()，自动推进
        break;
      }

      // ---- 脚本任务：通过插件执行脚本，不挂起 ----
      case 'bpmn:scriptTask': {
        const script = node.properties?.script;
        if (script) {
          // 调用 ScriptExecutorPlugin 执行脚本
          const result = await ctx.executeScript(script);
          ctx.setTokenData(result);
        }
        // 不调 suspend()，引擎自动推进
        break;
      }

      default:
        // 未知类型，自动通过
        break;
    }
  },

  // --- 节点离开钩子 ---
  onNodeLeave: async (ctx) => {
    // 如果是 userTask，标记任务完成
    if (ctx.node.type === 'bpmn:userTask') {
      await db.tasks.update(
        { processId: ctx.state.id, nodeId: ctx.node.id },
        { status: 'completed', completedAt: new Date() }
      );
    }
  },

  // --- 连线钩子 ---
  onSequenceFlow: (ctx) => {
    // 审计日志
    console.log(
      `[AUDIT] ${ctx.sourceNode.name} ──[${ctx.flow.id}]──> ${ctx.targetNode.name}`
    );
  },

  // --- 流程结束钩子 ---
  onProcessEnd: async (ctx) => {
    await db.processStates.update(ctx.state.id, ctx.state);
    await notifyService.send('admin', { title: `流程 ${ctx.state.name} 已完成` });
  },
});

// ========== 启动流程 ==========
const bpmnXML = fs.readFileSync('leave-request.bpmn', 'utf-8');

let state = await engine.startProcess(bpmnXML, {
  applicant: '张三',
  days: 5,
  reason: '年假',
});

// 持久化
await db.processStates.save(state);
// 同时保存 bpmnXML，resume 时需要

// 查看挂起的任务
const pending = engine.getSuspendedTokens(state, bpmnXML);
console.log(pending);
// [{ tokenId: 'abc', nodeType: 'bpmn:userTask', nodeName: '主管审批', ... }]


// ========== 前端审批接口 ==========
app.post('/api/approve/:processId/:tokenId', async (req, res) => {
  const { processId, tokenId } = req.params;
  const { approved, comment } = req.body;

  // 从 DB 加载 state 和 xml
  let state = await db.processStates.get(processId);
  const bpmnXML = await db.bpmnDefinitions.get(state.definitionId);

  // 恢复令牌
  state = await engine.resume(state, tokenId, bpmnXML, {
    approved,
    comment,
    approver: req.user.name,
    approvedAt: new Date(),
  });

  // 持久化新状态
  await db.processStates.update(processId, state);

  res.json({ ok: true, status: state.status });
});
```

```typescript
describe('SimpleFlow: start → userTask → end', () => {
  it('should suspend at userTask and resume to completion', async () => {
    const engine = new FlowEngine();
    const suspendedNodes: string[] = [];

    // 注册钩子：userTask 挂起
    engine.on('nodeEnter', (ctx) => {
      if (ctx.node.type === 'bpmn:userTask') {
        ctx.suspend();
        suspendedNodes.push(ctx.node.id);
      }
    });

    // 启动
    let state = await engine.startProcess(SIMPLE_BPMN_XML);

    // 验证：流程 running，1 个 suspended token
    expect(state.status).toBe('running');
    expect(state.tokens).toHaveLength(1);
    expect(state.tokens[0].status).toBe('suspended');

    // 恢复
    state = await engine.resume(
      state,
      state.tokens[0].id,
      SIMPLE_BPMN_XML,
      { approved: true }
    );

    // 验证：流程完成
    expect(state.status).toBe('completed');
    expect(state.tokens).toHaveLength(0);
    expect(suspendedNodes).toEqual(['userTask_1']);

    // 验证 trace
    const nodeEnters = state.trace.filter(t => t.type === 'node-enter');
    expect(nodeEnters.map(t => t.elementType)).toEqual([
      'bpmn:startEvent',
      'bpmn:userTask',
      'bpmn:endEvent',
    ]);
  });
});
```
